🔧 Keep-alive SQL for NON-HMG-product projects

School Connect / Tutoring Connect / DramaConnect / HMG CBT ship sc_keep_alive already. For ANY other Supabase project, paste this ONCE in its Supabase → SQL Editor and it becomes fully monitorable by this console. It creates one tiny heartbeat table + one SECURITY DEFINER function; RLS on your real tables is untouched.

-- HMG Fleet Console — keep-alive endpoint (safe to run twice)
create table if not exists public.sc_keepalive(
  id int primary key,
  pinged_at timestamptz,
  src text
);
alter table public.sc_keepalive enable row level security;
drop policy if exists sc_keepalive_read on public.sc_keepalive;
create policy sc_keepalive_read on public.sc_keepalive
  for select using (true);   -- heartbeat age only; contains no business data

create or replace function public.sc_keep_alive(src text default 'fleet')
returns timestamptz language sql security definer set search_path = public as $$
  insert into public.sc_keepalive(id, pinged_at, src) values (1, now(), left(coalesce(src,'fleet'),40))
  on conflict (id) do update set pinged_at = now(), src = excluded.src
  returning pinged_at;
$$;
grant execute on function public.sc_keep_alive(text) to anon, authenticated;
